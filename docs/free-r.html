Module 3: Data Visualization

3.1 Your First Bar Chart

# Simple data
months <- c("Jan", "Feb", "Mar", "Apr", "May")
sales <- c(15000, 18000, 22000, 19000, 21000)

# Create bar chart
barplot(sales, names.arg = months,
        main = "Monthly Sales",
        xlab = "Month",
        ylab = "Sales ($)",
        col = "steelblue")

3.2 Adding More Details

# Create a more detailed chart
png("sales_chart.png", width = 800, height = 600)

barplot(sales, names.arg = months,
        main = "Monthly Sales Report",
        xlab = "Month",
        ylab = "Sales ($)",
        col = "steelblue",
        ylim = c(0, max(sales) * 1.2))

# Add horizontal line for average
abline(h = mean(sales), col = "red", lty = 2, lwd = 2)
legend("topleft", legend = "Average", col = "red", lty = 2)

dev.off()
print("Chart saved as sales_chart.png")
[1] "Chart saved as sales_chart.png"

Exercise 2: Temperature Analysis

# Weekly temperatures (Fahrenheit)
temps <- c(72, 75, 78, 76, 73, 70, 75)
days <- c("Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun")

# Analysis
avg_temp <- mean(temps)
print(paste("Average Temperature:", round(avg_temp, 1), "°F"))

# Which days were above average?
above_avg <- temps > avg_temp
print(paste("Days above average:", paste(days[above_avg], collapse = ", ")))

# Visualize
barplot(temps, names.arg = days, col = "coral", main = "Weekly Temperature")
abline(h = avg_temp, col = "darkred", lwd = 2)

Exercise 3: Sales by Region

# Create comprehensive sales data
regions <- c("North", "South", "East", "West")
sales_by_region <- c(45000, 52000, 38000, 41000)

# Analysis
total <- sum(sales_by_region)
percentages <- round((sales_by_region / total) * 100, 1)

# Create data frame
region_df <- data.frame(
  Region = regions,
  Sales = sales_by_region,
  Percentage = percentages
)

print(region_df)

# Pie chart
pie(sales_by_region, labels = regions,
    main = "Sales Distribution by Region")

Project: Quarterly Business Review

# Quarterly performance data
quarters <- c("Q1", "Q2", "Q3", "Q4")
revenue <- c(150000, 175000, 200000, 185000)
expenses <- c(120000, 135000, 150000, 145000)
profit <- revenue - expenses

# Create summary data frame
qbr <- data.frame(
  Quarter = quarters,
  Revenue = revenue,
  Expenses = expenses,
  Profit = profit,
  Margin = round((profit / revenue) * 100, 1)
)

print(qbr)

# Visualization
par(mfrow = c(1, 2))

# Revenue vs Expenses
plot(revenue, type = "o", col = "blue", lwd = 2,
     main = "Revenue & Expenses", xlab = "Quarter", ylab = "Amount ($)")
lines(expenses, type = "o", col = "red", lwd = 2)
legend("topleft", legend = c("Revenue", "Expenses"), col = c("blue", "red"), lty = 1)

# Profit margin trend
plot(qbr$Margin, type = "o", col = "green", lwd = 2,
     main = "Profit Margin Trend", xlab = "Quarter", ylab = "Margin (%)")
abline(h = mean(qbr$Margin), col = "gray", lty = 2)

par(mfrow = c(1, 1))
