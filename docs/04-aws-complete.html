import json

template = {
    "AWSTemplateFormatVersion": "2010-09-09",

            "Properties": {
                "CidrBlock": "10.0.0.0/16"

    }

    StackName='my-vpc-stack',
    TemplateBody=json.dumps(template)
)

# Lambda Function (Python)
def lambda_handler(event, context):
    """Process S3 events"""

# Architecture Components
class AWSStack:
    def __init__(self):
        self.ec2 = boto3.client('ec2')
        self.s3 = boto3.client('s3')
        self.rds = boto3.client('rds')
        self.elbv2 = boto3.client('elbv2')

    def create_vpc_infrastructure(self):
        # VPC creation
        vpc = self.ec2.create_vpc(CidrBlock='10.0.0.0/16')
        vpc_id = vpc['Vpc']['VpcId']

        # Subnets
        public_subnet = self.ec2.create_subnet(
            VpcId=vpc_id,
            CidrBlock='10.0.1.0/24',
            AvailabilityZone='us-east-1a'
        )

        private_subnet = self.ec2.create_subnet(
            VpcId=vpc_id,
            CidrBlock='10.0.2.0/24',
            AvailabilityZone='us-east-1b'
        )

        return vpc_id, public_subnet['Subnet']['SubnetId'], private_subnet['Subnet']['SubnetId']

    def create_load_balancer(self, vpc_id, subnet_ids):
        response = self.elbv2.create_load_balancer(
            Name='my-alb',
            Subnets=subnet_ids,
            SecurityGroups=['sg-12345678'],
            Scheme='internet-facing'
        )
        return response['LoadBalancers'][0]['LoadBalancerArn']

Knowledge Check (3 Questions):

  1. Question 1: What is the maximum file size that can be uploaded to S3 using the standard put_object API?

      1. 1 MB
      1. 5 GB
      1. 100 MB
      1. 5 TB

    Answer: B Explanation: S3 has a 5 GB limit per put_object request. Larger files require multipart upload.

  2. Question 2: Which AWS service is best for real-time streaming data analytics?

      1. S3
      1. Redshift
      1. Kinesis
      1. Athena

    Answer: C Explanation: Kinesis handles real-time data streaming. Redshift is for batch analytics, Athena for S3 queries, S3 for storage.

  3. Question 3: What is the default retention period for CloudTrail logs in S3?

      1. 7 days
      1. 30 days
      1. 90 days
      1. No default limit (indefinite)

    Answer: D Explanation: CloudTrail delivers logs to S3 but doesn’t delete them. Retention is controlled by S3 lifecycle policies.
