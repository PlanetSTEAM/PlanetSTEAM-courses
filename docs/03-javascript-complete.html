// Async/Await Pattern
async function fetchUserData(userId) {
  try {
    const response = await fetch(`/api/users/${userId}`);
    if (!response.ok) throw new Error("User not found");

    const user = await response.json();
    return user;
  } catch (error) {
    console.error("Fetch error:", error);
    throw error;
  }
}

// Promises Chain
fetch('/api/data')
  .then(response => response.json())
  .then(data => console.log("Success:", data))
  .catch(error => console.error("Error:", error))
  .finally(() => console.log("Request complete"));

// Functional Programming Example
const numbers = [1, 2, 3, 4, 5];

const result = numbers
  .filter(n => n > 2)        // [3, 4, 5]
  .map(n => n * 2)           // [6, 8, 10]
  .reduce((sum, n) => sum + n, 0); // 24

// Higher-order functions
const compose = (...fns) => x =>
  fns.reduceRight((acc, fn) => fn(acc), x);

const add10 = x => x + 10;
const multiply2 = x => x * 2;

const calculate = compose(multiply2, add10);
console.log(calculate(5)); // (5 + 10) * 2 = 30

// Closures Example
function createCounter() {
  let count = 0;

  return {
    increment: () => ++count,
    decrement: () => --count,
    getCount: () => count
  };
}

const counter = createCounter();
console.log(counter.increment()); // 1
console.log(counter.increment()); // 2
console.log(counter.getCount());  // 2

Knowledge Check (3 Questions):

  1. Question 1: What is the difference between var, let, and const?

      1. No difference, all are the same
      1. var is function-scoped; let/const are block-scoped; const cannot be reassigned
      1. const is always global
      1. let and const are identical

    Answer: B Explanation: var has function scope and is hoisted. let and const have block scope. const prevents reassignment but allows property mutations for objects.

  2. Question 2: What will this code output?

    async function test() {
  console.log(1);
  await Promise.resolve();
  console.log(2);
}

test();
console.log(3);
      1. 1, 2, 3
      1. 1, 3, 2
      1. 3, 1, 2
      1. 2, 1, 3

    Answer: B Explanation: 1 is logged, then test() awaits a Promise (pushed to microtask). 3 is logged immediately. Then 2 is logged when Promise resolves.

  3. Question 3: What is the output?

    const numbers = [1, 2, 3, 4, 5];
const result = numbers
  .filter(n => n > 2)
  .map(n => n * 2)
  .reduce((sum, n) => sum + n, 0);
console.log(result);
      1. 15
      1. 24
      1. 12
      1. 30

    Answer: B Explanation: Filter: [3,4,5], Map: [6,8,10], Reduce: 6+8+10=24
