Module 2: Pandas Fundamentals

2.1 Creating DataFrames

import pandas as pd

# Sample data
data = {
    'Month': ['January', 'February', 'March', 'April'],
    'Sales': [15000, 18000, 22000, 19000],
    'Region': ['North', 'South', 'East', 'West']
}

df = pd.DataFrame(data)
print(df)
      Month  Sales Region
0   January  15000  North
1  February  18000  South
2     March  22000   East
3     April  19000   West

2.2 Basic Analysis

# Summary statistics
print(f"Average Sales: ${df['Sales'].mean():,.0f}")
print(f"Total Sales: ${df['Sales'].sum():,.0f}")
print(f"Highest Month: {df.loc[df['Sales'].idxmax(), 'Month']}")
Average Sales: $18,500
Total Sales: $74,000
Highest Month: March

Module 3: Data Visualization

3.1 Your First Chart

import matplotlib.pyplot as plt

# Create a bar chart
plt.figure(figsize=(10, 6))
plt.bar(df['Month'], df['Sales'], color='steelblue')
plt.xlabel('Month')
plt.ylabel('Sales ($)')
plt.title('Monthly Sales by Region')
plt.xticks(rotation=45)
plt.grid(axis='y', alpha=0.3)
plt.tight_layout()
plt.show()

3.2 Adding More Insights

# Grouped visualization
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))

# Bar chart
ax1.bar(df['Month'], df['Sales'], color='steelblue')
ax1.set_title('Monthly Sales')
ax1.set_ylabel('Sales ($)')

# Pie chart
ax2.pie(df['Sales'], labels=df['Month'], autopct='%1.1f%%')
ax2.set_title('Sales Distribution')

plt.tight_layout()
plt.show()

Project: Sales Analysis Dashboard

Create a complete analysis of quarterly sales data:

# Quarterly sales data
quarters = ['Q1', 'Q2', 'Q3', 'Q4']
sales = [150000, 175000, 200000, 185000]
targets = [140000, 160000, 190000, 180000]

# Create DataFrame
performance = pd.DataFrame({
    'Quarter': quarters,
    'Sales': sales,
    'Target': targets
})

# Calculate performance
performance['Performance %'] = (performance['Sales'] / performance['Target'] * 100).round(1)
performance['Gap'] = performance['Sales'] - performance['Target']

print(performance)

# Visualize
fig, ax = plt.subplots(figsize=(10, 6))
x = range(len(quarters))
width = 0.35

ax.bar([i - width/2 for i in x], sales, width, label='Sales', color='steelblue')
ax.bar([i + width/2 for i in x], targets, width, label='Target', color='coral')
ax.set_xlabel('Quarter')
ax.set_ylabel('Amount ($)')
ax.set_title('Quarterly Sales vs Target')
ax.set_xticks(x)
ax.set_xticklabels(quarters)
ax.legend()
ax.grid(axis='y', alpha=0.3)
plt.tight_layout()
plt.show()
